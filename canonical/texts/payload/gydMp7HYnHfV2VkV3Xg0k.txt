Avec l'applet suivant, vous pouvez examiner les deux demi-périodes séparément.<br><br>

<html>
    <script src="https://www.50ohm.de/assets/jquery/3.7.1/jquery.min.js"></script>
    <script src="https://www.50ohm.de/assets/chartjs/4.4.1/chart.min.js"></script>

    <button id="button0">Aucune action</button>
    <button id="button1">1ère demi-période</button>
    <button id="button2">2ème demi-période</button>

    <img id="GR1" width="100%"/>
    
    <script>
        const image = [
            "https://50ohm.de/pictures/962.svg",
            "https://50ohm.de/pictures/963.svg",
            "https://50ohm.de/pictures/964.svg",
        ];

        document.getElementById('button0').addEventListener('click', function() {
            cycle(0);
        });

        document.getElementById('button1').addEventListener('click', function() {
            cycle(1);
        });

        document.getElementById('button2').addEventListener('click', function() {
            cycle(2);
        });

        function cycle(index = 0) {
            document.querySelector('#GR1').src = image[index];
        }

        cycle(0);

    </script>
</html>
